// AJAX handler for getting clinic details with related data
function ajax_get_clinic_details() {
    check_ajax_referer('clinic_nonce', 'nonce');

    $clinic_id = absint($_POST['clinic_id'] ?? 0);
    
    $clinic = $this->aiclinic_get_clinic_by_id($clinic_id);
    if (!$clinic) {
        wp_send_json_error(['message' => 'Clinic not found']);
    }

    // Get tags
    $clinic['tags'] = $this->aiclinic_get_tags_by_clinic($clinic_id);

    // Get reviews
    $reviews = $this->aiclinic_get_reviews_by_clinic($clinic_id, ['limit' => 5]);
    $clinic['recent_reviews'] = $reviews;

    // Get statistics
    $clinic['avg_rating'] = $this->aiclinic_get_clinic_average_rating($clinic_id);
    $clinic['review_count'] = $this->aiclinic_get_clinic_review_count($clinic_id);

    wp_send_json_success($clinic);
}

// AJAX handler for bulk operations
function ajax_bulk_delete_clinics() {
    check_ajax_referer('clinic_nonce', 'nonce');

    if (!current_user_can('manage_options')) {
        wp_send_json_error(['message' => 'Permission denied']);
    }

    $clinic_ids = array_map('absint', $_POST['clinic_ids'] ?? []);
    
    if (empty($clinic_ids)) {
        wp_send_json_error(['message' => 'No clinics selected']);
    }

    $deleted = $this->aiclinic_bulk_delete_clinics($clinic_ids);

    wp_send_json_success([
        'message' => "Deleted $deleted clinics",
        'deleted' => $deleted
    ]);
}

// Advanced search with multiple filters
function advanced_clinic_search($params) {
    $args = [
        'search'         => $params['keyword'] ?? '',
        'column_search'  => 'name',
        'limit'          => $params['per_page'] ?? 20,
        'offset'         => ($params['page'] ?? 0) * ($params['per_page'] ?? 20),
        'order_by'       => $params['order_by'] ?? 'created_at',
        'sort_direction' => $params['sort'] ?? 'DESC'
    ];

    // Add filters
    if (!empty($params['province'])) {
        $args['province'] = $params['province'];
    }

    if (!empty($params['district'])) {
        $args['district'] = $params['district'];
    }

    if (!empty($params['verify_status'])) {
        $args['verify_status'] = $params['verify_status'];
    }

    if (!empty($params['is_active'])) {
        $args['is_active'] = $params['is_active'];
    }

    return $this->aiclinic_get_clinics_paginated($args);
}

// Search clinics by tag
function search_clinics_by_tags($tag_names, $limit = 20) {
    // Get tag IDs
    $tag_ids = [];
    foreach ($tag_names as $name) {
        $tag = $this->aiclinic_get_tag_by_name($name);
        if ($tag) {
            $tag_ids[] = $tag['id'];
        }
    }

    if (empty($tag_ids)) {
        return [];

    // Get clinics for first tag
    $clinics = $this->aiclinic_get_clinics_by_tag($tag_ids[0], ['limit' => $limit]);

        $filtered = [];
        foreach ($clinics as $clinic) {
            $clinic_tags = $this->aiclinic_get_tags_by_clinic($clinic['id']);
            $clinic_tag_ids = array_column($clinic_tags, 'id');
            
            $has_all_tags = true;
            foreach ($tag_ids as $tag_id) {
                if (!in_array($tag_id, $clinic_tag_ids)) {
                    $has_all_tags = false;
                    break;

    // ========== COMPLETE REGISTRATION WORKFLOW ==========
    function register_new_customer($data) {
        // Validate domain
        if ($this->aiclinic_domain_exists($data['domain'])) {
            return ['success' => false, 'message' => 'Domain already registered'];
        }

        // Generate customer key
        $customer_key = $this->aiclinic_generate_customer_key();

        // Create customer
        $customer_id = $this->aiclinic_insert_customer([
            'user_id'         => get_current_user_id(),
            'customer_key'    => $customer_key,
            'customer_name'   => sanitize_text_field($data['name']),
            'pricing_plan'    => $data['plan'] ?? 'BASIC',
            'domain'          => sanitize_text_field($data['domain']),
            'customer_status' => 'UNPAID'
        ]);

        if (!$customer_id) {
            return ['success' => false, 'message' => 'Failed to create customer'];
        }

        // Send welcome email
        $this->send_welcome_email($customer_id, $customer_key);

        return [
            'success'      => true,
            'customer_id'  => $customer_id,
            'customer_key' => $customer_key,
            'message'      => 'Registration successful'
        ];
    }

    // ========== UPGRADE CUSTOMER PLAN ==========
    function upgrade_customer_to_gold($customer_id) {
        // Get customer
        $customer = $this->aiclinic_get_customer_by_id($customer_id);
        
        if (!$customer) {
            return false;
        }

        // Update plan
        $updated = $this->aiclinic_update_customer_plan($customer_id, 'GOLD');

        if ($updated) {
            // Log the upgrade
            $this->aiclinic_insert_error_log([
                'action'  => 'customer_upgrade',
                'message' => "Customer upgraded to GOLD plan",
                'context' => [
                    'customer_id'   => $customer_id,
                    'old_plan'      => $customer['pricing_plan'],
                    'new_plan'      => 'GOLD',
                    'upgraded_by'   => get_current_user_id()
                ]
            ]);

            // Send notification

    // ========== AJAX: Update Customer Status ==========
    function ajax_update_customer_status() {
        check_ajax_referer('customer_nonce', 'nonce');

        if (!current_user_can('manage_options')) {
            wp_send_json_error(['message' => 'Permission denied']);
        }

        $customer_id = absint($_POST['customer_id'] ?? 0);
        $new_status = sanitize_text_field($_POST['status'] ?? '');

        if (!in_array($new_status, ['ACTIVE', 'INACTIVE', 'UNPAID'])) {
            wp_send_json_error(['message' => 'Invalid status']);
        }

        $result = $this->aiclinic_update_customer_status($customer_id, $new_status);

        if ($result) {
            // Log the change
            $this->aiclinic_insert_error_log([
                'action'  => 'customer_status_change',
                'message' => "Customer status updated to $new_status",
                'context' => [
                    'customer_id' => $customer_id,
                    'new_status'  => $new_status,
                    'updated_by'  => get_current_user_id()
                ]
            ]);

            wp_send_json_success(['message' => 'Status updated successfully']);
        } else {
            wp_send_json_error(['message' => 'Failed to update status']);
        }
    }

    // ========== AJAX: Bulk Action on Customers ==========
    function ajax_bulk_customer_action() {
        check_ajax_referer('customer_nonce', 'nonce');

        if (!current_user_can('manage_options')) {
            wp_send_json_error(['message' => 'Permission denied']);
        }

        $action = sanitize_text_field($_POST['action'] ?? '');
        $customer_ids = array_map('absint', $_POST['customer_ids'] ?? []);

        if (empty($customer_ids)) {
            wp_send_json_error(['message' => 'No customers selected']);
        }

        switch ($action) {
            case 'delete':
                $deleted = $this->aiclinic_bulk_delete_customers($customer_ids);
                wp_send_json_success(['message' => "Deleted $deleted customers"]);
                break;

            case 'activate':
                $updated = 0;
                foreach ($customer_ids as $id) {
                    if ($this->aiclinic_update_customer_status($id, 'ACTIVE')) {
                        $updated++;
                    }
                }
                wp_send_json_success(['message' => "Activated $updated customers"]);
                break;

            case 'deactivate':
                $updated = 0;
                foreach ($customer_ids as $id) {
                    if ($this->aiclinic_update_customer_status($id, 'INACTIVE')) {
                        $updated++;
                    }
                }
                wp_send_json_success(['message' => "Deactivated $updated customers"]);
                break;

            default:
                wp_send_json_error(['message' => 'Invalid action']);
        }
    }
